import heapq
def astar(graph,h,start,goal):
    pq=[(0,start,[start])]
    g={start:0}
    while pq:
        f,node,path=heapq.heappop(pq)
        if node==goal:
            return path,g[node]
        for n,cost in graph[node]:
            new=g[node]+cost
        if n not in g or new<g[n]:
            g[n]=new
            heapq.heappush(pq,(new+h[n],n,path+[n]))
graph={
    'A':[('B',1),('C',2)],
    'B':[('D',1)],
    'C':[('E',2),('D',3)]
}
h={'A':1,'B':2,'C':3,'D':4,'E':5}
path,cost=(astar(graph,h,'A','D'))
print(path)
print(cost)
