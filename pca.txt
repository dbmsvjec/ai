from sklearn.decomposition import PCA
import numpy as np

# Example dataset: rows = images, columns = features
data = np.array([
    [2.5, 2.4, 1.2],
    [0.5, 0.7, 0.3],
    [2.2, 2.9, 1.5],
    [1.9, 2.2, 1.1],
    [3.1, 3.0, 1.7]
])

# Reduce features to 2 principal components
pca = PCA(n_components=2)

reduced_data = pca.fit_transform(data)

print("Reduced Data:\n", reduced_data)

print("\nPrincipal Components:\n", pca.components_)

print("\nExplained Variance:\n", pca.explained_variance_ratio_)
